# Mine - the game on top of Tile2D. Only built when the renderer is available.
#
# mine_core is deliberately free of any graphics dependency (and of any game content): it holds the
# session description and the start screen model, so it is testable in milliseconds and reusable by
# a dedicated server later. The tests link it directly.
#
# The game itself is **not linked into the executable**: it is compiled into a code table that the
# launcher merges with whatever mods are beside it (docs/TABLES.md).

add_library(mine_core STATIC
    src/content_grid.cpp
    src/content_list.cpp
    src/content_loader.cpp
    src/content_pack.cpp
    src/content_search.cpp
    src/layer_rules.cpp
    src/menu.cpp
    src/mod_package.cpp
    src/registry.cpp
    src/sandbox.cpp
    src/types/entity_tile.cpp
    src/types/floor.cpp
    src/types/scene_tile.cpp
    src/types/tile.cpp
    src/types/tile_definition.cpp
    src/world.cpp
)
add_library(mine::core ALIAS mine_core)
t2d_apply_settings(mine_core)
target_include_directories(mine_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(mine_core PUBLIC t2d::core)

# What a native mod compiles against: the API header and the types it names, and nothing to link.
# A mod is a separate shared library with its own copy of whatever it uses, so this is an include-only
# target on purpose - the one thing a mod must never do is link the game's static libraries.
add_library(mine_mod_sdk INTERFACE)
target_include_directories(mine_mod_sdk INTERFACE
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_SOURCE_DIR}/include"
    "${PROJECT_BINARY_DIR}/generated")   # t2d/config.h, generated by the framework's configure step

# ------------------------------------------------------------- the game, as a code table ----------
# Everything the game is - its logic and its shell - is compiled into **one table** instead of being
# linked into the executable. The flag that matters is -fsemantic-interposition: it keeps a call
# inside the game a call *through a relocation*, which is what lets a mod table merged after it
# replace a function of the game and have the game's own calls follow.
#
# The table is compiled by **the toolchain**, not by this build system, and that is the point: what the
# toolchain compiles with *is* the ABI, and a table it builds carries the record of it (docs/ABI.md).
# It also links what it compiled - a partial link, `-r` - before packing it: the game is one program,
# and a program has one copy of an inline function rather than one per translation unit (docs/TABLES.md §2).
# The engine's own record is written by the same tool (engine.abi, below), so "was this mod built for
# this engine" is answered by what the compiler said rather than by a version string.
set(MINE_TABLE_NAME "mine.codetab")
set(MINE_TABLE_SOURCES
    src/app.cpp
    src/content_grid.cpp
    # The entry point the launcher would call, packed with everything else so that the table is a whole
    # program rather than a library: it is here, and not in the executable, on purpose.
    src/game_main.cpp
    src/content_list.cpp
    src/content_loader.cpp
    src/content_pack.cpp
    src/content_search.cpp
    src/layer_rules.cpp
    src/menu.cpp
    src/mod_package.cpp
    src/registry.cpp
    src/sandbox.cpp
    src/types/entity_tile.cpp
    src/types/floor.cpp
    src/types/scene_tile.cpp
    src/types/tile.cpp
    src/types/tile_definition.cpp
    src/world.cpp
)
# What the toolchain is told about this build. The includes and the defines are the ones the sources
# need to compile at all; the optimisation follows the build type and is deliberately **not** part of
# the ABI - a mod built at -O0 and an engine built at -O3 agree on every layout fact and are merged
# like one program (docs/ABI.md §0).
set(MINE_TABLE_INCLUDES
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_SOURCE_DIR}/include"
    "${PROJECT_BINARY_DIR}/generated"
    "${T2D_ORE_DIR}/include"
    "${CMAKE_BINARY_DIR}/ore/generated")
set(MINE_TABLE_DEFINES
    "T2D_SOURCE_DIR=\"${CMAKE_SOURCE_DIR}\""
    "ORE_SOURCE_DIR=\"${T2D_ORE_DIR}\""
    "ORE_SOURCE_SHADER_DIR=\"${T2D_ORE_DIR}/shaders\"")
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(MINE_TABLE_OPT "-O0")
elseif(CMAKE_BUILD_TYPE STREQUAL "Release")
    set(MINE_TABLE_OPT "-O3")
else()
    set(MINE_TABLE_OPT "-O2")
endif()
if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    list(APPEND MINE_TABLE_DEFINES "NDEBUG")
endif()
set(MINE_TOOLCHAIN_ARGS --opt ${MINE_TABLE_OPT})
foreach(include ${MINE_TABLE_INCLUDES})
    list(APPEND MINE_TOOLCHAIN_ARGS --include "${include}")
endforeach()
foreach(define ${MINE_TABLE_DEFINES})
    list(APPEND MINE_TOOLCHAIN_ARGS --define "${define}")
endforeach()

set(MINE_TABLE_SOURCES_ABS "")
foreach(source ${MINE_TABLE_SOURCES})
    list(APPEND MINE_TABLE_SOURCES_ABS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
endforeach()
set(MINE_TABLE_PATH "${CMAKE_CURRENT_BINARY_DIR}/${MINE_TABLE_NAME}")
add_custom_command(
    OUTPUT "${MINE_TABLE_PATH}"
    COMMAND $<TARGET_FILE:codetab> build ${MINE_TABLE_SOURCES_ABS}
            -o "${MINE_TABLE_PATH}" --id mine --name "Mine" --version 1.0 --requires engine@1.0
            --api "${CMAKE_SOURCE_DIR}/engine.api" ${MINE_TOOLCHAIN_ARGS}
    DEPENDS codetab ${MINE_TABLE_SOURCES_ABS}
    # The source list is a list, not one long argument: without this the shell sees the semicolons.
    COMMAND_EXPAND_LISTS
    # A define like T2D_SOURCE_DIR="/path" carries quotes the shell must not eat.
    VERBATIM
    COMMENT "Building the game into ${MINE_TABLE_NAME} (and recording what it was built as)")
add_custom_target(mine_game_table ALL DEPENDS "${MINE_TABLE_PATH}")

# The one source the toolchain compiles that nothing else here does, compiled here too so that -Werror
# covers it. Nothing links these objects: the table is what runs.
add_library(mine_entry_check OBJECT src/game_main.cpp)
t2d_apply_settings(mine_entry_check)
target_include_directories(mine_entry_check PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(mine_entry_check PRIVATE t2d::render ore::ore)

# What the **launcher** itself was compiled as, written by the same tool that records a table. It
# travels with the executable the way engine.api does, and it is the reference every module is
# measured against: a table built against other headers, or against another C++ library ABI, is
# refused with the fact that differs named - never loaded "with fewer rights" (docs/ABI.md).
set(MINE_ABI_PATH "${CMAKE_CURRENT_BINARY_DIR}/engine.abi")
add_custom_command(
    OUTPUT "${MINE_ABI_PATH}"
    COMMAND $<TARGET_FILE:codetab> abi --record "${MINE_ABI_PATH}" ${MINE_TOOLCHAIN_ARGS}
    DEPENDS codetab
    COMMAND_EXPAND_LISTS
    VERBATIM
    COMMENT "Writing the engine's own ABI record (${MINE_ABI_PATH})")
add_custom_target(mine_engine_abi ALL DEPENDS "${MINE_ABI_PATH}")

# The published surface travels with the executable: the launcher reads it to decide what a module may
# ask the engine for, and which versions it may be from.
add_custom_command(TARGET mine_game_table POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_SOURCE_DIR}/engine.api" "${CMAKE_CURRENT_BINARY_DIR}/engine.api"
    COMMENT "Copying engine.api beside the game")

# A demo mod: one table that replaces a function of the game's own table. It is built, not installed:
# the game only loads it when it is asked to (--table), so a plain run stays a plain run. The toolchain
# builds it, so it carries its own ABI record and is measured against the engine's like any other mod.
set(MINE_DEMO_MOD "${CMAKE_CURRENT_BINARY_DIR}/mods/demo_mod/mod.codetab")
add_custom_command(
    OUTPUT "${MINE_DEMO_MOD}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/mods/demo_mod"
    COMMAND $<TARGET_FILE:codetab> build "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/tables/mod_banner.cpp"
            -o "${MINE_DEMO_MOD}" --id demo_mod --version 1.0 --requires mine@1.0
            --api "${CMAKE_SOURCE_DIR}/engine.api"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/tables/mod_banner.cpp" codetab
    COMMENT "Building the demo mod's code table")
add_custom_target(mine_demo_mod ALL DEPENDS "${MINE_DEMO_MOD}")

# The app shell, linked normally. The executable does not use it any more - the game arrives as the
# table above and the executable is a launcher (src/main.cpp) - but the tests do: they link the shell
# and drive the screens directly, which is a millisecond a case instead of a window and a GPU.
add_library(mine_app STATIC
    src/app.cpp
)
add_library(mine::app ALIAS mine_app)
t2d_apply_settings(mine_app)
target_link_libraries(mine_app PUBLIC mine_core t2d::render)

add_executable(mine_game src/main.cpp)
t2d_apply_settings(mine_game)
# The engine is linked whole, and that is not an optimisation detail: the game arrives as a table and
# resolves the engine's symbols at run time, so every object file of it has to be in this program's
# dynamic symbol table - including the ones this program never calls itself. A static library linked
# normally only keeps what the linker was asked for, which would leave the game unable to find the
# framework's own functions.
target_link_libraries(mine_game PRIVATE
    -Wl,--whole-archive t2d::render t2d::sim t2d::text t2d::net t2d::core ore::ore -Wl,--no-whole-archive)
set_target_properties(mine_game PROPERTIES ENABLE_EXPORTS ON)
add_dependencies(mine_game mine_game_table mine_engine_abi)
ore_add_shaders(mine_game SHADERS
    "${CMAKE_SOURCE_DIR}/shaders/sprite.vert"
    "${CMAKE_SOURCE_DIR}/shaders/sprite.frag")

# The game's own content pack travels beside the executable, so a build tree is a runnable game and
# the source tree is not consulted at run time at all: what is packaged is what was built.
add_custom_command(TARGET mine_game_table POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_SOURCE_DIR}/games/mine/content" "${CMAKE_CURRENT_BINARY_DIR}/content"
    COMMENT "Copying the game's own content pack beside the executable")

# A release: everything a run needs, assembled in one directory and then zipped. Nothing here is
# optional - a package without the table, the surface or the content is a package that does not start.
# Debug information belongs to the build tree, where somebody is actually debugging: a package that
# carries nine megabytes of it is a package that is nine megabytes too large for everyone else. The
# switch is here for the one case that wants it - handing somebody a binary to debug.
option(MINE_PACKAGE_DEBUG_SYMBOLS "Keep debug information in the package" OFF)

set(MINE_PACKAGE_NAME "mine-${PROJECT_VERSION}")
set(MINE_PACKAGE_ROOT "${CMAKE_BINARY_DIR}/package")
set(MINE_PACKAGE_DIR "${MINE_PACKAGE_ROOT}/${MINE_PACKAGE_NAME}")
set(MINE_PACKAGE_ZIP "${CMAKE_BINARY_DIR}/${MINE_PACKAGE_NAME}-${CMAKE_BUILD_TYPE}.zip")
# The working directory of a custom command has to exist before the command runs, and the commands
# below are what put the package together: the root is made at configure time, the rest at build time.
file(MAKE_DIRECTORY "${MINE_PACKAGE_ROOT}")
set(MINE_PACKAGE_STRIP "")
if(NOT MINE_PACKAGE_DEBUG_SYMBOLS)
    # --strip-all drops the symbol table and the debug sections; the *dynamic* symbols stay, which is
    # what the launcher exports and the game's table resolves the engine through. Stripping a launcher
    # whose symbols the game needs is exactly the mistake this comment exists to prevent.
    set(MINE_PACKAGE_STRIP COMMAND ${CMAKE_STRIP} --strip-all "${MINE_PACKAGE_DIR}/mine_game")
endif()

add_custom_target(mine_package
    COMMAND ${CMAKE_COMMAND} -E remove_directory "${MINE_PACKAGE_DIR}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${MINE_PACKAGE_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:mine_game> "${MINE_PACKAGE_DIR}/"
    COMMAND ${CMAKE_COMMAND} -E copy "${MINE_TABLE_PATH}" "${MINE_PACKAGE_DIR}/"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/engine.api" "${MINE_PACKAGE_DIR}/"
    # What the launcher was built as: the reference a mod's own record is measured against.
    COMMAND ${CMAKE_COMMAND} -E copy "${MINE_ABI_PATH}" "${MINE_PACKAGE_DIR}/"
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_SOURCE_DIR}/games/mine/content" "${MINE_PACKAGE_DIR}/content"
    # The interface strings, where the game looks for them beside its executable. Without them a
    # package shows ids instead of text, and only a machine that still had the source tree would not.
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_SOURCE_DIR}/assets" "${MINE_PACKAGE_DIR}/assets"
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_CURRENT_BINARY_DIR}/shaders" "${MINE_PACKAGE_DIR}/shaders"
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_SOURCE_DIR}/packs" "${MINE_PACKAGE_DIR}/packs"
    ${MINE_PACKAGE_STRIP}
    COMMAND ${CMAKE_COMMAND} -E remove "${MINE_PACKAGE_ZIP}"
    COMMAND ${CMAKE_COMMAND} -E tar cf "${MINE_PACKAGE_ZIP}" --format=zip -- "${MINE_PACKAGE_NAME}"
    WORKING_DIRECTORY "${MINE_PACKAGE_ROOT}"
    DEPENDS mine_game mine_game_table mine_engine_abi
    COMMENT "Packaging ${MINE_PACKAGE_NAME} into ${MINE_PACKAGE_ZIP}")

# A package for the people who write mods, which is a different thing from a package for the people
# who play: the headers the engine and the game are made of, the generated configuration headers they
# include, the published surface, the game's own table (so a mod can be built and checked against the
# game it mods), the toolchain, the template and the documents that explain all of it.
set(MINE_DEV_NAME "mine-dev-${PROJECT_VERSION}")
set(MINE_DEV_DIR "${MINE_PACKAGE_ROOT}/${MINE_DEV_NAME}")
set(MINE_DEV_ZIP "${CMAKE_BINARY_DIR}/${MINE_DEV_NAME}-${CMAKE_BUILD_TYPE}.zip")
add_custom_target(builddev
    COMMAND ${CMAKE_COMMAND} -E remove_directory "${MINE_DEV_DIR}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${MINE_DEV_DIR}/include"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${T2D_ORE_DIR}/include/ore"
            "${MINE_DEV_DIR}/include/ore"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/include/t2d"
            "${MINE_DEV_DIR}/include/t2d"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/games/mine/include/mine"
            "${MINE_DEV_DIR}/include/mine"
    # The headers that are generated rather than written: t2d/config.h, which every t2d header includes,
    # and ore/config.h, which ore's headers include - a package without either does not compile.
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${PROJECT_BINARY_DIR}/generated/t2d"
            "${MINE_DEV_DIR}/include/t2d"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_BINARY_DIR}/ore/generated/ore"
            "${MINE_DEV_DIR}/include/ore"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/engine.api" "${MINE_DEV_DIR}/engine.api"
    # The engine's own record, so a mod author can diff what they built against what they build for.
    COMMAND ${CMAKE_COMMAND} -E copy "${MINE_ABI_PATH}" "${MINE_DEV_DIR}/engine.abi"
    COMMAND ${CMAKE_COMMAND} -E copy "${MINE_TABLE_PATH}" "${MINE_DEV_DIR}/mine.codetab"
    COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:codetab> "${MINE_DEV_DIR}/codetab"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/packs" "${MINE_DEV_DIR}/packs"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/docs" "${MINE_DEV_DIR}/docs"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/README.md" "${MINE_DEV_DIR}/README.md"
    COMMAND ${CMAKE_COMMAND} -E remove "${MINE_DEV_ZIP}"
    COMMAND ${CMAKE_COMMAND} -E tar cf "${MINE_DEV_ZIP}" --format=zip -- "${MINE_DEV_NAME}"
    WORKING_DIRECTORY "${MINE_PACKAGE_ROOT}"
    DEPENDS mine_game_table codetab
    COMMENT "Packaging ${MINE_DEV_NAME} into ${MINE_DEV_ZIP}")

if(T2D_BUILD_TESTS)
    add_subdirectory(tests)
endif()
