# Mefemma - a game project on Tile2D. Only built when the renderer is available.
#
# mefemma_core is deliberately free of any graphics dependency: settings, input, the draw list, the scene
# stack and the paths that find the game's files. It is testable in milliseconds and could run on a server.
# mefemma_app is the half that talks to the GPU: assets, the 2D renderer, the application shell and the two
# scenes that ship as examples.
#
# The game itself is **not linked into the executable**: it is compiled into a code table that the launcher
# merges with whatever mods are beside it (docs/TABLES.md).

add_library(mefemma_core STATIC
    src/data_paths.cpp
    src/draw_list.cpp
    src/input.cpp
    src/scene.cpp
    src/settings.cpp
)
add_library(mefemma::core ALIAS mefemma_core)
t2d_apply_settings(mefemma_core)
target_include_directories(mefemma_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
# t2d::text is CPU work only: scene.h hands a scene the string table and the text metrics, so the game's
# screens can lay text out without a GPU.
target_link_libraries(mefemma_core PUBLIC t2d::core t2d::text)

# What a mod compiles against: the headers and the types they name, and nothing to link. A mod is a separate
# shared library with its own copy of whatever it uses, so this is an include-only target on purpose - the
# one thing a mod must never do is link the game's static libraries.
add_library(mefemma_mod_sdk INTERFACE)
target_include_directories(mefemma_mod_sdk INTERFACE
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_SOURCE_DIR}/include"
    "${PROJECT_BINARY_DIR}/generated")   # t2d/config.h, generated by the framework's configure step

# The GPU half: pictures, the 2D renderer, the application shell and the scenes.
add_library(mefemma_app STATIC
    src/app.cpp
    src/assets.cpp
    src/renderer2d.cpp
    src/scenes/playground_scene.cpp
    src/scenes/title_scene.cpp
)
add_library(mefemma::app ALIAS mefemma_app)
t2d_apply_settings(mefemma_app)
target_link_libraries(mefemma_app PUBLIC mefemma_core t2d::render ore::ore)

# ------------------------------------------------------------- the game, as a code table ----------
# Everything the game is - its logic, its shell and its two example scenes - is compiled into **one table**
# instead of being linked into the executable. The flag that matters is -fsemantic-interposition: it keeps a
# call inside the game a call through a relocation, which is what lets a mod table merged after it replace a
# function of the game and have the game's own calls follow.
#
# The table is compiled by **the toolchain**, not by this build system, and that is the point: what the
# toolchain compiles with *is* the ABI, and a table it builds carries the record of it (docs/ABI.md).
set(MEFEMMA_TABLE_NAME "mefemma.codetab")
set(MEFEMMA_TABLE_SOURCES
    # The entry point the launcher calls, packed with everything else so that the table is a whole program
    # rather than a library: it is here, and not in the executable, on purpose.
    src/game_main.cpp
    src/app.cpp
    src/assets.cpp
    src/renderer2d.cpp
    src/scenes/playground_scene.cpp
    src/scenes/title_scene.cpp
    src/data_paths.cpp
    src/draw_list.cpp
    src/input.cpp
    src/scene.cpp
    src/settings.cpp
)
# What the toolchain is told about this build. The includes and the defines are the ones the sources need to
# compile at all; the optimisation follows the build type and is deliberately **not** part of the ABI - a mod
# built at -O0 and an engine built at -O3 agree on every layout fact and are merged like one program.
set(MEFEMMA_TABLE_INCLUDES
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_SOURCE_DIR}/include"
    "${PROJECT_BINARY_DIR}/generated"
    "${T2D_ORE_DIR}/include"
    "${CMAKE_BINARY_DIR}/ore/generated")
set(MEFEMMA_TABLE_DEFINES
    "T2D_SOURCE_DIR=\"${CMAKE_SOURCE_DIR}\""
    "ORE_SOURCE_DIR=\"${T2D_ORE_DIR}\""
    "ORE_SOURCE_SHADER_DIR=\"${T2D_ORE_DIR}/shaders\"")
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(MEFEMMA_TABLE_OPT "-O0")
elseif(CMAKE_BUILD_TYPE STREQUAL "Release")
    set(MEFEMMA_TABLE_OPT "-O3")
else()
    set(MEFEMMA_TABLE_OPT "-O2")
endif()
if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    list(APPEND MEFEMMA_TABLE_DEFINES "NDEBUG")
endif()
set(MEFEMMA_TOOLCHAIN_ARGS --opt ${MEFEMMA_TABLE_OPT})
foreach(include ${MEFEMMA_TABLE_INCLUDES})
    list(APPEND MEFEMMA_TOOLCHAIN_ARGS --include "${include}")
endforeach()
foreach(define ${MEFEMMA_TABLE_DEFINES})
    list(APPEND MEFEMMA_TOOLCHAIN_ARGS --define "${define}")
endforeach()

set(MEFEMMA_TABLE_SOURCES_ABS "")
foreach(source ${MEFEMMA_TABLE_SOURCES})
    list(APPEND MEFEMMA_TABLE_SOURCES_ABS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
endforeach()
set(MEFEMMA_TABLE_PATH "${CMAKE_CURRENT_BINARY_DIR}/${MEFEMMA_TABLE_NAME}")
# What the table is built **from** is more than its sources: a header it includes is part of what it is. A
# header changed under a table that was not rebuilt is a table that runs the old layout, so the headers are
# dependencies too. CONFIGURE_DEPENDS, so a header that is new is picked up by a rebuild.
file(GLOB_RECURSE MEFEMMA_TABLE_HEADERS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h"
    "${CMAKE_SOURCE_DIR}/include/*.h"
    "${PROJECT_BINARY_DIR}/generated/*.h")
add_custom_command(
    OUTPUT "${MEFEMMA_TABLE_PATH}"
    COMMAND $<TARGET_FILE:codetab> build ${MEFEMMA_TABLE_SOURCES_ABS}
            -o "${MEFEMMA_TABLE_PATH}" --id mefemma --name "Mefemma" --version 1.0 --requires engine@1.0
            --api "${CMAKE_SOURCE_DIR}/engine.api" ${MEFEMMA_TOOLCHAIN_ARGS}
    DEPENDS codetab ${MEFEMMA_TABLE_SOURCES_ABS} ${MEFEMMA_TABLE_HEADERS} "${CMAKE_SOURCE_DIR}/engine.api"
    # The source list is a list, not one long argument: without this the shell sees the semicolons.
    COMMAND_EXPAND_LISTS
    # A define like T2D_SOURCE_DIR="/path" carries quotes the shell must not eat.
    VERBATIM
    COMMENT "Building the game into ${MEFEMMA_TABLE_NAME} (and recording what it was built as)")
add_custom_target(mefemma_game_table ALL DEPENDS "${MEFEMMA_TABLE_PATH}")

# A demo mod: one table that replaces a function of the game's own table. It is built, not installed: the
# game only loads it when it is asked to (--table), so a plain run stays a plain run. The toolchain builds
# it, so it carries its own ABI record and is measured against the engine's like any other mod.
set(MEFEMMA_DEMO_MOD "${CMAKE_CURRENT_BINARY_DIR}/mods/demo_mod/mod.codetab")
add_custom_command(
    OUTPUT "${MEFEMMA_DEMO_MOD}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/mods/demo_mod"
    COMMAND $<TARGET_FILE:codetab> build "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/tables/mod_banner.cpp"
            -o "${MEFEMMA_DEMO_MOD}" --id mefemma_demo --version 1.0 --requires engine@1.0
            --api "${CMAKE_SOURCE_DIR}/engine.api" ${MEFEMMA_TOOLCHAIN_ARGS}
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/tables/mod_banner.cpp" codetab
    COMMENT "Building the demo mod's code table")
add_custom_target(mefemma_demo_mod ALL DEPENDS "${MEFEMMA_DEMO_MOD}")

# The one source the toolchain compiles that nothing else here does, compiled here too so that -Werror
# covers it. Nothing links these objects: the table is what runs.
add_library(mefemma_entry_check OBJECT src/game_main.cpp)
t2d_apply_settings(mefemma_entry_check)
target_include_directories(mefemma_entry_check PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(mefemma_entry_check PRIVATE mefemma_app)

# What the **launcher** itself was compiled as, written by the same tool that records a table. It travels
# with the executable the way engine.api does, and it is the reference every module is measured against.
set(MEFEMMA_ABI_PATH "${CMAKE_CURRENT_BINARY_DIR}/engine.abi")
add_custom_command(
    OUTPUT "${MEFEMMA_ABI_PATH}"
    COMMAND $<TARGET_FILE:codetab> abi --record "${MEFEMMA_ABI_PATH}" ${MEFEMMA_TOOLCHAIN_ARGS}
    DEPENDS codetab
    COMMAND_EXPAND_LISTS
    VERBATIM
    COMMENT "Writing the engine's own ABI record (${MEFEMMA_ABI_PATH})")
add_custom_target(mefemma_engine_abi ALL DEPENDS "${MEFEMMA_ABI_PATH}")

# The published surface travels with the executable: the launcher reads it to decide what a module may ask
# the engine for, and which versions it may be from.
add_custom_command(TARGET mefemma_game_table POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_SOURCE_DIR}/engine.api" "${CMAKE_CURRENT_BINARY_DIR}/engine.api"
    COMMENT "Copying engine.api beside the game")

add_executable(mefemma_game src/main.cpp)
t2d_apply_settings(mefemma_game)
# The engine is linked whole, and that is not an optimisation detail: the game arrives as a table and
# resolves the engine's symbols at run time, so every object file of it has to be in this program's dynamic
# symbol table - including the ones this program never calls itself. A static library linked normally only
# keeps what the linker was asked for, which would leave the game unable to find the framework's own
# functions.
target_link_libraries(mefemma_game PRIVATE
    -Wl,--whole-archive t2d::render t2d::sim t2d::text t2d::net t2d::core ore::ore -Wl,--no-whole-archive)
set_target_properties(mefemma_game PROPERTIES ENABLE_EXPORTS ON)
add_dependencies(mefemma_game mefemma_game_table mefemma_engine_abi)
ore_add_shaders(mefemma_game SHADERS
    "${CMAKE_SOURCE_DIR}/shaders/sprite.vert"
    "${CMAKE_SOURCE_DIR}/shaders/sprite.frag")

# The assets travel beside the executable, so a build tree is a runnable game: the settings, the strings and
# the pictures are read from what was built, not from the source tree. (The source tree is the fallback for
# a bare executable - DataPaths::resolve says when.)
add_custom_command(TARGET mefemma_game_table POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_SOURCE_DIR}/games/mefemma/assets" "${CMAKE_CURRENT_BINARY_DIR}/assets"
    COMMENT "Copying the game's assets beside the executable")

# A release: everything a run needs, assembled in one directory and then zipped. Nothing here is optional -
# a package without the table, the surface, the strings or the pictures is a package that does not start.
option(MEFEMMA_PACKAGE_DEBUG_SYMBOLS "Keep debug information in the package" OFF)

set(MEFEMMA_PACKAGE_NAME "mefemma-${PROJECT_VERSION}")
set(MEFEMMA_PACKAGE_ROOT "${CMAKE_BINARY_DIR}/package")
set(MEFEMMA_PACKAGE_DIR "${MEFEMMA_PACKAGE_ROOT}/${MEFEMMA_PACKAGE_NAME}")
set(MEFEMMA_PACKAGE_ZIP "${CMAKE_BINARY_DIR}/${MEFEMMA_PACKAGE_NAME}-${CMAKE_BUILD_TYPE}.zip")
file(MAKE_DIRECTORY "${MEFEMMA_PACKAGE_ROOT}")
set(MEFEMMA_PACKAGE_STRIP "")
if(NOT MEFEMMA_PACKAGE_DEBUG_SYMBOLS)
    # --strip-all drops the symbol table and the debug sections; the *dynamic* symbols stay, which is what
    # the launcher exports and the game's table resolves the engine through.
    set(MEFEMMA_PACKAGE_STRIP COMMAND ${CMAKE_STRIP} --strip-all "${MEFEMMA_PACKAGE_DIR}/mefemma_game")
endif()

add_custom_target(mefemma_package
    COMMAND ${CMAKE_COMMAND} -E remove_directory "${MEFEMMA_PACKAGE_DIR}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${MEFEMMA_PACKAGE_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:mefemma_game> "${MEFEMMA_PACKAGE_DIR}/"
    COMMAND ${CMAKE_COMMAND} -E copy "${MEFEMMA_TABLE_PATH}" "${MEFEMMA_PACKAGE_DIR}/"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/engine.api" "${MEFEMMA_PACKAGE_DIR}/"
    # What the launcher was built as: the reference a mod's own record is measured against.
    COMMAND ${CMAKE_COMMAND} -E copy "${MEFEMMA_ABI_PATH}" "${MEFEMMA_PACKAGE_DIR}/"
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_SOURCE_DIR}/games/mefemma/assets" "${MEFEMMA_PACKAGE_DIR}/assets"
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_CURRENT_BINARY_DIR}/shaders" "${MEFEMMA_PACKAGE_DIR}/shaders"
    ${MEFEMMA_PACKAGE_STRIP}
    COMMAND ${CMAKE_COMMAND} -E remove "${MEFEMMA_PACKAGE_ZIP}"
    COMMAND ${CMAKE_COMMAND} -E tar cf "${MEFEMMA_PACKAGE_ZIP}" --format=zip -- "${MEFEMMA_PACKAGE_NAME}"
    WORKING_DIRECTORY "${MEFEMMA_PACKAGE_ROOT}"
    DEPENDS mefemma_game mefemma_game_table mefemma_engine_abi
    COMMENT "Packaging ${MEFEMMA_PACKAGE_NAME} into ${MEFEMMA_PACKAGE_ZIP}")

if(T2D_BUILD_TESTS)
    add_subdirectory(tests)
endif()
